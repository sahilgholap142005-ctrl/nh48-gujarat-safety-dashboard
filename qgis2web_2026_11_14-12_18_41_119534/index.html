// Store both charts in global variables so filter events can update them var mechanismChartInstance = new Chart(document.getElementById('mechanismChart'), { type: 'bar', data: { labels: ['Rear-End', 'Sideswipe', 'Head-On', 'Run-Off', 'Parked'], datasets: [{ data: [978, 740, 476, 265, 185], backgroundColor: ['#f59e0b', '#8b5cf6', '#ef4444', '#10b981', '#3b82f6'], borderRadius: 4 }] }, options: { plugins: { legend: { display: false } }, scales: { x: { ticks: { color: '#cbd5e1', font: { size: 10 } }, grid: { display: false } }, y: { ticks: { color: '#94a3b8', font: { size: 10 } }, grid: { color: '#334155' } } } } }); var districtChartInstance = new Chart(document.getElementById('districtChart'), { type: 'doughnut', data: { labels: ['Surat Rural', 'Bharuch', 'Vadodara', 'Valsad', 'Anand/Other'], datasets: [{ data: [34, 24, 19, 13, 10], backgroundColor: ['#ef4444', '#f97316', '#eab308', '#38bdf8', '#64748b'], borderColor: '#1e293b', borderWidth: 2 }] }, options: { plugins: { legend: { position: 'right', labels: { color: '#e2e8f0', font: { size: 11 }, boxWidth: 12 } } } } }); // Dynamically recalculate Chart.js graphs whenever a qgis2web filter changes function syncChartsWithMapFilters() { // Default baseline counts (used when all filters are cleared) var mechCounts = { 'Rear-End': 0, 'Sideswipe': 0, 'Head-On': 0, 'Run-Off': 0, 'Parked': 0 }; var distCounts = { 'Surat Rural': 0, 'Bharuch': 0, 'Vadodara': 0, 'Valsad': 0, 'Anand/Other': 0 }; var visibleFeatureCount = 0; // Scan all currently active vector point layers on the Leaflet map map.eachLayer(function (layer) { if (layer.feature && layer.feature.properties) { var p = layer.feature.properties; // Weight by CLUSTER_SIZE if reading blackspot centroids, otherwise 1 per crash var weight = p.CLUSTER_SIZE ? Number(p.CLUSTER_SIZE) : 1; visibleFeatureCount += weight; // 1. Tally Collision Mechanism var nature = String(p.C_Collision_Nature || '').toLowerCase(); if (nature.includes('rear')) mechCounts['Rear-End'] += weight; else if (nature.includes('sideswipe') || nature.includes('side')) mechCounts['Sideswipe'] += weight; else if (nature.includes('head')) mechCounts['Head-On'] += weight; else if (nature.includes('run') || nature.includes('overturn') || nature.includes('skid')) mechCounts['Run-Off'] += weight; else if (nature.includes('parked') || nature.includes('stationary')) mechCounts['Parked'] += weight; // 2. Tally District Breakdown var dist = String(p.C_District || p.District || '').toLowerCase(); if (dist.includes('surat')) distCounts['Surat Rural'] += weight; else if (dist.includes('bharuch')) distCounts['Bharuch'] += weight; else if (dist.includes('vadodara')) distCounts['Vadodara'] += weight; else if (dist.includes('valsad')) distCounts['Valsad'] += weight; else if (dist) distCounts['Anand/Other'] += weight; } }); // Update charts only if active vector markers exist on screen if (visibleFeatureCount > 0) { mechanismChartInstance.data.datasets[0].data = [ mechCounts['Rear-End'], mechCounts['Sideswipe'], mechCounts['Head-On'], mechCounts['Run-Off'], mechCounts['Parked'] ]; mechanismChartInstance.update(); districtChartInstance.data.datasets[0].data = [ distCounts['Surat Rural'], distCounts['Bharuch'], distCounts['Vadodara'], distCounts['Valsad'], distCounts['Anand/Other'] ]; districtChartInstance.update(); } } // Listen for clicks, dropdown selections, and slider releases inside qgis2web's #menu var filterMenu = document.getElementById('menu'); if (filterMenu) { ['change', 'click', 'mouseup'].forEach(function (evt) { filterMenu.addEventListener(evt, function () { // Wait 120ms for qgis2web's filterFunc() to finish adding/removing markers setTimeout(syncChartsWithMapFilters, 120); }); }); } // Auto-clean legend and filter text artifacts on load window.addEventListener('load', function () { setTimeout(function () { // 1. Fix "-1000000000000" and "♦" characters inside the Leaflet Layer Control document.querySelectorAll('.leaflet-control-layers-overlays td, .leaflet-control-layers-overlays label').forEach(function (el) { el.innerHTML = el.innerHTML .replace(/-1000000000000/g, '130') .replace(/\uFFFD/g, ' - ') .replace(/♦/g, ' - '); }); // 2. Clean filter dropdown options (remove "NaN", fix diamonds and typo) document.querySelectorAll('#menu select option').forEach(function (opt) { if (opt.textContent.trim() === 'NaN') { opt.remove(); } else { opt.textContent = opt.textContent .replace(/\uFFFD/g, ' - ') .replace(/♦/g, ' - ') .replace('Gnevous', 'Grievous'); } }); }, 400); });
